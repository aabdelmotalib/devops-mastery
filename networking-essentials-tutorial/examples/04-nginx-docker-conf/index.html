    nginx-docker.conf - Development Reverse Proxy Configuration

    Lightweight Nginx configuration for development and Docker environments, demonstrating reverse proxy setup for load balancing and SSL termination.

    user nginx;
worker_processes auto;
error_log /var/log/nginx/error.log warn;
pid /var/run/nginx.pid;

events {
    worker_connections 1024;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;

    log_format main '$remote_addr - $remote_user [$time_local] "$request" '
                    '$status $body_bytes_sent "$http_referer" '
                    '"$http_user_agent" "$http_x_forwarded_for"';

    access_log /var/log/nginx/access.log main;

    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;
    types_hash_max_size 2048;
    client_max_body_size 20M;

    gzip on;
    gzip_vary on;
    gzip_types text/plain text/css application/json application/javascript;
    gzip_comp_level 6;

    # Backend upstream
    upstream backend {
        least_conn;
        server app1:5001 max_fails=3 fail_timeout=30s;
        server app2:5002 max_fails=3 fail_timeout=30s;
        server app3:5003 max_fails=3 fail_timeout=30s;
        server app4:5004 max_fails=3 fail_timeout=30s;
    }

    # Rate limiting
    limit_req_zone $binary_remote_addr zone=api_limit:10m rate=100r/m;
    limit_req_status 429;

    # Cache
    proxy_cache_path /var/cache/nginx levels=1:2 keys_zone=api_cache:100m max_size=10g inactive=60m;

    # HTTP redirect to HTTPS
    server {
        listen 80;
        server_name _;

        location / {
            return 301 https://$host$request_uri;
        }

        location /health {
            access_log off;
            return 200 "OK\n";
            add_header Content-Type text/plain;
        }
    }

    # HTTPS (for testing with self-signed cert)
    server {
        listen 443 ssl http2 default_server;
        server_name _;

        # Self-signed cert (for testing only)
        ssl_certificate /etc/nginx/certs/cert.pem;
        ssl_certificate_key /etc/nginx/certs/key.pem;

        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers HIGH:!aNULL:!MD5;
        ssl_prefer_server_ciphers on;

        # Security headers
        add_header Strict-Transport-Security "max-age=31536000" always;
        add_header X-Content-Type-Options "nosniff" always;
        add_header X-Frame-Options "DENY" always;

        # Load balancer health check
        location /lb-health {
            access_log off;
            return 200 "OK\n";
            add_header Content-Type text/plain;
        }

        # Health check endpoint
        location /health {
            access_log off;
            proxy_pass http://backend;
        }

        # API with rate limiting and caching
        location /api/ {
            limit_req zone=api_limit burst=20 nodelay;

            proxy_pass http://backend;
            proxy_http_version 1.1;
            proxy_set_header Connection "";

            # Headers
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            # Caching
            proxy_cache api_cache;
            proxy_cache_valid 200 5m;
            add_header X-Cache-Status $upstream_cache_status;

            # Timeouts
            proxy_connect_timeout 60s;
            proxy_send_timeout 60s;
            proxy_read_timeout 60s;
        }

        # Default location
        location / {
            proxy_pass http://backend;
            proxy_http_version 1.1;
            proxy_set_header Connection "";

            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }
    }

    # Status page
    server {
        listen 8080;
        server_name _;

        location /nginx_status {
            stub_status on;
            access_log off;
            allow 127.0.0.1;
            allow 172.16.0.0/12;  # Docker network
            deny all;
        }
    }
}

    Configuration Breakdown

    Upstream Backend Definition

    upstream backend {
    least_conn;
    server app1:5001 max_fails=3 fail_timeout=30s;
    server app2:5002 max_fails=3 fail_timeout=30s;
    server app3:5003 max_fails=3 fail_timeout=30s;
    server app4:5004 max_fails=3 fail_timeout=30s;
}

    Explanation: - least_conn: Load balancing algorithm - routes to server with fewest connections - max_fails=3: Server marked unhealthy after 3 failed requests - fail_timeout=30s: Server excluded from load balancing for 30 seconds - app1:5001 etc: Service names in Docker network with internal ports

    HTTP to HTTPS Redirect

    server {
    listen 80;
    server_name _;

    location / {
        return 301 https://$host$request_uri;
    }
}

    Redirects all HTTP traffic to HTTPS (301 permanent redirect).

    SSL/TLS Configuration

    ssl_certificate /etc/nginx/certs/cert.pem;
ssl_certificate_key /etc/nginx/certs/key.pem;

ssl_protocols TLSv1.2 TLSv1.3;
ssl_ciphers HIGH:!aNULL:!MD5;
ssl_prefer_server_ciphers on;

    Development-only setup with self-signed certificates.

    Rate Limiting

    limit_req_zone $binary_remote_addr zone=api_limit:10m rate=100r/m;
limit_req zone=api_limit burst=20 nodelay;
    • 10m: Shared memory zone size
    • 100r/m: 100 requests per minute per IP
    • burst=20: Allow 20 requests above limit
    • nodelay: Don't delay, return 429 immediately

    Caching

    proxy_cache_path /var/cache/nginx levels=1:2 keys_zone=api_cache:100m max_size=10g inactive=60m;

proxy_cache api_cache;
proxy_cache_valid 200 5m;
    • Cache HTTP 200 responses for 5 minutes
    • X-Cache-Status header shows cache hit/miss

    Usage in Docker Compose

    Mount this configuration into Nginx container:

    nginx:
  image: nginx:alpine
  volumes:
    - ./examples/nginx-docker.conf:/etc/nginx/nginx.conf:ro
    - ./examples/certs:/etc/nginx/certs:ro
  ports:
    - "80:80"
    - "443:443"
  depends_on:
    - app1
    - app2
    - app3
    - app4

    Testing

    Validate Configuration

    # Inside container
docker-compose exec nginx nginx -t

# Or from host
docker run --rm -v $(pwd)/nginx-docker.conf:/etc/nginx/nginx.conf:ro nginx:alpine nginx -t

    Test Load Balancing

    # Make multiple requests and see different app servers
for i in {1..10}; do
  curl http://localhost/api/v1/health
done

# Check logs to see distribution
docker-compose logs nginx | grep "upstream:"

    Test Caching

    # First request - cache miss
curl -i http://localhost/api/v1/products | grep X-Cache-Status

# Second request - cache hit
curl -i http://localhost/api/v1/products | grep X-Cache-Status

    Test Rate Limiting

    # Make 25 requests quickly
for i in {1..25}; do
  curl http://localhost/api/v1/products
done

# After 100 requests/minute, should get 429

    Monitor Nginx Statistics

    # Access internal status page
curl http://localhost:8080/nginx_status

    Network Concepts Demonstrated

    1. Service Discovery

    • Nginx resolves app1, app2, etc. via Docker DNS
    • Service names are hostnames in the backend network

    2. Load Balancing

    • Round-robin (default): Cycles through servers
    • Least conn: Routes to server with fewest connections
    • IP hash: Routes based on client IP
    • Random: Random selection

    3. Health Checks

    • Failed requests mark server as unhealthy
    • Nginx removes unhealthy server temporarily
    • Server re-enabled after fail_timeout

    4. Connection Pooling

    proxy_http_version 1.1;
proxy_set_header Connection "";

    Keeps connections alive to upstream servers for performance.

    5. Header Forwarding

    proxy_set_header X-Real-IP $remote_addr;
proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
proxy_set_header X-Forwarded-Proto $scheme;

    Application sees original client IP, not Nginx's IP.

    6. Reverse Proxy

    Nginx hides backend servers: - Clients connect to Nginx on port 80/443 - Nginx forwards to internal backend on various ports - Backends not exposed to clients

    Security Considerations

    Development Only

    • Self-signed certificate (not valid for production)
    • No authentication on /nginx_status
    • Default error pages

    For Production

    • Replace with real certificates (Let's Encrypt)
    • Restrict /nginx_status access
    • Add authentication
    • Use stronger cipher suites
    • Enable HSTS

    Connection Flow Example

    Client HTTP request to localhost:80
  ↓
Nginx receives on port 80
  ↓
Returns 301 redirect to https://localhost
  ↓
Client makes HTTPS request to localhost:443
  ↓
Nginx receives HTTPS, decrypts with cert.pem
  ↓
Nginx looks up "upstream backend"
  ↓
Nginx picks app1 (least_conn algorithm)
  ↓
Nginx connects to app1:5001 in docker network
  ↓
Nginx forwards request, reads response
  ↓
Nginx encrypts response with cert.pem
  ↓
Nginx sends HTTPS response to client
  ↓
Client receives response (TLS termination complete)

    Logs

    Access logs show format: 

    172.19.0.1 - - [24/Feb/2026:10:30:45 +0000] "GET /api/v1/products HTTP/1.1" 200 1234 "-" "curl/7.85.0"

    Check logs: 

    docker-compose logs -f nginx
docker-compose logs nginx --tail 50

    Troubleshooting

    Backend Services Not Found

    # Check Docker network
docker network inspect backend

# Verify app containers are running
docker-compose ps

# Test DNS from Nginx
docker-compose exec nginx nslookup app1

    502 Bad Gateway

    # Check if backend is healthy
docker-compose logs app1

# Test connection from Nginx
docker-compose exec nginx curl http://app1:5001/health

    SSL Certificate Errors

    # Verify certificate exists
ls -la examples/certs/

# Check certificate validity
openssl x509 -in examples/certs/cert.pem -text -noout

    Rate Limiting Not Working

    # Verify rate limit zone is configured
docker-compose exec nginx grep "limit_req_zone" /etc/nginx/nginx.conf

# Test with slow requests
ab -n 150 -c 1 http://localhost/api/v1/products
