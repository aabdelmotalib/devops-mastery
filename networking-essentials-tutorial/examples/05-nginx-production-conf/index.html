    nginx-production.conf - Production Reverse Proxy Configuration

    Production-grade Nginx configuration with advanced features including strong TLS, security headers, caching, rate limiting, and WebSocket support.

    # Nginx configuration for reverse proxy, load balancing, and TLS termination
# This is a production-ready configuration demonstrating all networking concepts

# Upstream backend definition (load balancing targets)
upstream backend {
    # Use least connections algorithm
    # Other options: round_robin (default), ip_hash, etc.
    least_conn;

    # Backend servers with health check configuration
    server 127.0.0.1:5001 max_fails=3 fail_timeout=30s;
    server 127.0.0.1:5002 max_fails=3 fail_timeout=30s;
    server 127.0.0.1:5003 max_fails=3 fail_timeout=30s;
    server 127.0.0.1:5004 max_fails=3 fail_timeout=30s;

    # max_fails: disable server after N failed attempts
    # fail_timeout: disable for this duration before retrying
}

# Cache definition
proxy_cache_path /var/cache/nginx levels=1:2 keys_zone=api_cache:100m max_size=10g inactive=60m;

# Rate limiting zones (in http block, before server block)
limit_req_zone $binary_remote_addr zone=general_limit:10m rate=100r/m;
limit_req_zone $binary_remote_addr zone=api_limit:10m rate=500r/m;
limit_req_zone $http_x_api_key zone=api_key_limit:10m rate=5000r/m;

# Redirect HTTP to HTTPS
server {
    listen 80;
    listen [::]:80;
    server_name example.com www.example.com;

    # Allow Let's Encrypt verification
    location /.well-known/acme-challenge/ {
        root /var/www/certbot;
    }

    # Redirect all other traffic to HTTPS
    location / {
        return 301 https://$server_name$request_uri;
    }
}

# Main HTTPS server
server {
    # Listen on HTTPS with HTTP/2 support
    listen 443 ssl http2;
    listen [::]:443 ssl http2;

    server_name example.com www.example.com;

    # SSL/TLS configuration
    ssl_certificate /etc/letsencrypt/live/example.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/example.com/privkey.pem;

    # Use only TLS 1.2 and 1.3 (disable older protocols)
    ssl_protocols TLSv1.2 TLSv1.3;

    # Strong cipher suites
    ssl_ciphers 'ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:DHE-RSA-AES128-GCM-SHA256:DHE-RSA-AES256-GCM-SHA384';
    ssl_prefer_server_ciphers on;

    # Enable SSL session caching
    ssl_session_cache shared:SSL:50m;
    ssl_session_timeout 1d;

    # HSTS (Strict-Transport-Security)
    # Tells browsers to always use HTTPS for future requests
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains; preload" always;

    # Security headers
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Frame-Options "DENY" always;
    add_header X-XSS-Protection "1; mode=block" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;

    # Gzip compression
    gzip on;
    gzip_vary on;
    gzip_types text/plain text/css application/json application/javascript text/xml application/xml application/xml+rss;
    gzip_comp_level 6;
    gzip_min_length 1000;

    # Logging
    access_log /var/log/nginx/access.log combined;
    error_log /var/log/nginx/error.log warn;

    # Health check endpoint (for monitoring)
    location /lb-health {
        access_log off;
        return 200 "healthy\n";
        add_header Content-Type text/plain;
    }

    # API endpoints with rate limiting
    location /api/ {
        # Apply rate limiting
        limit_req zone=api_limit burst=50 nodelay;

        # Check API key for higher limit
        if ($http_x_api_key ~* "^sk-") {
            limit_req zone=api_key_limit burst=200 nodelay;
        }

        # Caching configuration
        proxy_cache api_cache;
        proxy_cache_valid 200 5m;
        proxy_cache_bypass $http_cache_control;

        # Show cache status in response header
        add_header X-Cache-Status $upstream_cache_status;

        # Proxy settings
        proxy_pass http://backend;
        proxy_http_version 1.1;
        proxy_set_header Connection "";

        # Forward important headers
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $server_name;

        # Timeouts
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;

        # Buffering
        proxy_buffering on;
        proxy_buffer_size 4k;
        proxy_buffers 24 4k;
    }

    # Static files (images, CSS, JS) - longer cache
    location ~* \.(jpg|jpeg|png|gif|ico|css|js|svg|woff|woff2|ttf|eot)$ {
        proxy_pass http://backend;

        # Cache static assets for 30 days
        proxy_cache api_cache;
        proxy_cache_valid 200 30d;

        # Add cache control headers
        add_header Cache-Control "public, max-age=2592000, immutable";
        add_header X-Cache-Status $upstream_cache_status;

        # Forward headers
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # Long timeout for large files
        proxy_read_timeout 300s;
    }

    # WebSocket support (if needed)
    location /ws/ {
        proxy_pass http://backend;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # WebSocket timeout (keep connection open)
        proxy_read_timeout 3600s;
        proxy_send_timeout 3600s;
    }

    # Health check endpoint from backend
    location /health {
        access_log off;
        proxy_pass http://backend;
        proxy_set_header X-Real-IP $remote_addr;
    }

    # Admin panel - IP restricted
    location /admin/ {
        # Allow from localhost and office network
        allow 127.0.0.1;
        allow 203.0.113.0/24;    # Replace with your office IP range
        deny all;

        proxy_pass http://backend;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Deny access to sensitive files
    location ~ /\. {
        deny all;
        access_log off;
        log_not_found off;
    }

    # Root location
    location / {
        proxy_pass http://backend;

        # Forward headers
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # HTTP version
        proxy_http_version 1.1;
        proxy_set_header Connection "";
    }
}

# Status page (internal monitoring)
server {
    listen 127.0.0.1:8080;
    server_name _;

    location /nginx_status {
        stub_status on;
        access_log off;
        allow 127.0.0.1;
        deny all;
    }
}

    Production Features

    TLS/SSL Security

    Strong Encryption

    • TLSv1.2 and 1.3 only: No deprecated protocols
    • ECDHE key exchange: Perfect forward secrecy
    • Strong cipher suites: Modern encryption algorithms

    Certificate Management

    • Uses Let's Encrypt certificates
    • Automatic renewal via certbot
    • HTTP-01 challenge verification

    HSTS (HTTP Strict Transport Security)

    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains; preload" always;
    - Forces browsers to use HTTPS for 1 year - Prevents SSL stripping attacks - Submit to HSTS preload list for maximum security

    Security Headers

    X-Content-Type-Options: nosniff          # Prevents MIME type sniffing
X-Frame-Options: DENY                     # Prevents clickjacking
X-XSS-Protection: 1; mode=block          # Browser XSS protection
Referrer-Policy: strict-origin-when-cross-origin  # Privacy

    Advanced Load Balancing

    Upstream Configuration

    upstream backend {
    least_conn;  # Connection-based load balancing
    server 127.0.0.1:5001 max_fails=3 fail_timeout=30s;
}

    Algorithms: - round_robin (default): Cycles through servers - least_conn: Fewest connections - ip_hash: Client IP-based (sticky sessions) - least_time: Fastest response time (NGINX Plus)

    Health Checks

    • max_fails=3: Mark unhealthy after 3 failures
    • fail_timeout=30s: Retry after 30 seconds
    • Automatic removal of dead servers

    Caching Strategy

    Cache Configuration

    proxy_cache_path /var/cache/nginx levels=1:2 keys_zone=api_cache:100m max_size=10g inactive=60m;
    • levels=1:2: Directory structure for cache keys
    • keys_zone=api_cache:100m: Shared memory for fast lookups
    • max_size=10g: Maximum cache disk size
    • inactive=60m: Remove unused entries after 60 minutes

    Cache Policies

    # API endpoints: 5 minute cache
proxy_cache_valid 200 5m;

# Static files: 30 day cache
proxy_cache_valid 200 30d;

# Bypass cache if client sends Cache-Control
proxy_cache_bypass $http_cache_control;

    Rate Limiting

    Three-tier Strategy

    limit_req_zone $binary_remote_addr zone=general_limit:10m rate=100r/m;
limit_req_zone $binary_remote_addr zone=api_limit:10m rate=500r/m;
limit_req_zone $http_x_api_key zone=api_key_limit:10m rate=5000r/m;

    Rate Limiting Usage: - burst=50: Allow 50 requests above limit - nodelay: Return 429 immediately (don't queue) - Protects against brute force and DDoS

    WebSocket Support

    location /ws/ {
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "upgrade";
    proxy_read_timeout 3600s;  # Keep connection open
    proxy_send_timeout 3600s;
}

    Access Control

    IP Allowlisting

    location /admin/ {
    allow 127.0.0.1;
    allow 203.0.113.0/24;  # Office network
    deny all;
}

    Hidden Resources

    location ~ /\. {
    deny all;  # Block .git, .env, etc.
}

    Compression

    gzip on;
gzip_types text/plain text/css application/json application/javascript;
gzip_comp_level 6;  # Balance speed vs compression

    Reduces bandwidth for text-based responses.

    Deployment Setup

    Prerequisites

    1. Domain name pointed to server IP
    2. Certbot installed for SSL certificates
    3. Backend services running on ports 5001-5004

    Initial Setup

    # Install certbot
sudo apt-get install certbot python3-certbot-nginx

# Generate certificate (first time)
sudo certbot certonly --standalone -d example.com -d www.example.com

# Copy this config file
sudo cp nginx-production.conf /etc/nginx/sites-available/default

# Test configuration
sudo nginx -t

# Start nginx
sudo systemctl start nginx

# Enable auto-renewal
sudo systemctl enable nginx

    Monitoring & Logs

    Access Logs

    # Follow in real-time
tail -f /var/log/nginx/access.log

# Find slow requests
grep "HTTP/1" /var/log/nginx/access.log | awk '{print $(NF-1), $NF}' | sort -rn | head

    Internal Metrics

    # Access status page
curl http://localhost:8080/nginx_status

# Shows: active connections, accepts, handled, requests, reading, writing, waiting

    Health Monitoring

    # Check backend health
for port in 5001 5002 5003 5004; do
  curl -f http://127.0.0.1:$port/health && echo "Port $port OK" || echo "Port $port DOWN"
done

    Network Flows

    Request to API Endpoint

    Client HTTPS → Nginx:443 (TLS handshake)
  ↓
Nginx decrypts, checks X-API-Key rate limit
  ↓
Nginx looks up "upstream backend" (DNS)
  ↓
Nginx selects least_conn server (app1-4)
  ↓
Nginx connects to 127.0.0.1:5001
  ↓
Backend processes request
  ↓
Nginx caches response (if cacheable)
  ↓
Nginx encrypts with TLS
  ↓
Client receives HTTPS response

    Cache Hit vs Miss

    First request: Cache Miss
  Client → Nginx → Backend → Cache → Client

Second request (within 5m): Cache Hit
  Client → Nginx → Cached Response → Client
  (No backend connection needed)

    WebSocket Connection

    Client WS → Nginx:443 (TLS Upgrade)
  ↓
Nginx detects Upgrade header
  ↓
Nginx routes to /ws/ location
  ↓
Nginx connects to backend with Connection: Upgrade
  ↓
Nginx maintains persistent connection (3600s timeout)
  ↓
Bidirectional communication possible

    Troubleshooting

    Certificate Issues

    # Check certificate validity
openssl x509 -in /etc/letsencrypt/live/example.com/fullchain.pem -text

# Renew certificates
sudo certbot renew

# View certbot log
sudo tail -f /var/log/letsencrypt/letsencrypt.log

    Backend Unavailable

    # Check if backend is running
netstat -tlnp | grep 5001

# Test connectivity
curl http://127.0.0.1:5001/health

# Check Nginx error logs
sudo tail -f /var/log/nginx/error.log

    Rate Limiting Issues

    # Disable rate limiting temporarily for debugging
comment out limit_req lines
sudo nginx -s reload

# Test with ApacheBench
ab -n 1000 -c 10 https://example.com/api/

    Cache Not Working

    # Check cache directory
ls -la /var/cache/nginx/

# View cache keys
find /var/cache/nginx -type f | head

# Test cache header
curl -i https://example.com/api/products | grep X-Cache-Status

    SSL/TLS Handshake Issues

    # Test with OpenSSL
openssl s_client -connect example.com:443

# Check supported ciphers
openssl s_client -connect example.com:443 -cipher 'HIGH'

    Performance Tuning

    Upstream Keep-Alive

    keepalive 32;  # Keep 32 idle connections to backend
keepalive_requests 100;
keepalive_timeout 60s;

    Buffer Tuning

    proxy_buffer_size 4k;
proxy_buffers 24 4k;  # 96KB total for large responses
proxy_busy_buffers_size 8k;

    Connection Settings

    worker_connections 1024;  # Per worker process
worker_processes auto;  # CPU cores

    Zero-Downtime Deployment

    Graceful Reload

    # Load new configuration without dropping connections
sudo nginx -s reload

# Verify running
ps aux | grep nginx

    Health Check Integration

    # Remove unhealthy server from upstream
server 127.0.0.1:5001 down;
sudo nginx -s reload
