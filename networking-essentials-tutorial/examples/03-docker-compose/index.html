    Docker Compose - Multi-Service Orchestration

    Complete Docker Compose configuration demonstrating networking concepts including service-to-service communication, port mapping, volume management, and network configuration.

    version: '3.8'

services:
  # PostgreSQL Database
  postgres:
    image: postgres:15-alpine
    container_name: postgres
    environment:
      POSTGRES_USER: app
      POSTGRES_PASSWORD: app_password
      POSTGRES_DB: appdb
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U app -d appdb"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - backend

  # Redis Cache
  redis:
    image: redis:7-alpine
    container_name: redis
    ports:
      - "6379:6379"
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - backend

  # Backend Service 1
  app1:
    build:
      context: .
      dockerfile: Dockerfile
    container_name: app1
    environment:
      PORT: 5001
      INSTANCE_ID: app-1
      DATABASE_URL: postgresql://app:app_password@postgres:5432/appdb
      REDIS_URL: redis://redis:6379
      JWT_SECRET_KEY: your-secret-key-change-in-production
    ports:
      - "5001:5001"
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:5001/health"]
      interval: 10s
      timeout: 5s
      retries: 3
    networks:
      - backend

  # Backend Service 2
  app2:
    build:
      context: .
      dockerfile: Dockerfile
    container_name: app2
    environment:
      PORT: 5002
      INSTANCE_ID: app-2
      DATABASE_URL: postgresql://app:app_password@postgres:5432/appdb
      REDIS_URL: redis://redis:6379
      JWT_SECRET_KEY: your-secret-key-change-in-production
    ports:
      - "5002:5002"
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:5002/health"]
      interval: 10s
      timeout: 5s
      retries: 3
    networks:
      - backend

  # Backend Service 3
  app3:
    build:
      context: .
      dockerfile: Dockerfile
    container_name: app3
    environment:
      PORT: 5003
      INSTANCE_ID: app-3
      DATABASE_URL: postgresql://app:app_password@postgres:5432/appdb
      REDIS_URL: redis://redis:6379
      JWT_SECRET_KEY: your-secret-key-change-in-production
    ports:
      - "5003:5003"
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:5003/health"]
      interval: 10s
      timeout: 5s
      retries: 3
    networks:
      - backend

  # Backend Service 4
  app4:
    build:
      context: .
      dockerfile: Dockerfile
    container_name: app4
    environment:
      PORT: 5004
      INSTANCE_ID: app-4
      DATABASE_URL: postgresql://app:app_password@postgres:5432/appdb
      REDIS_URL: redis://redis:6379
      JWT_SECRET_KEY: your-secret-key-change-in-production
    ports:
      - "5004:5004"
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:5004/health"]
      interval: 10s
      timeout: 5s
      retries: 3
    networks:
      - backend

  # Nginx Reverse Proxy & Load Balancer
  nginx:
    image: nginx:alpine
    container_name: nginx
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./examples/nginx-docker.conf:/etc/nginx/nginx.conf:ro
      - ./ssl:/etc/nginx/ssl:ro
      # Self-signed cert for testing
      - ./examples/certs:/etc/nginx/certs:ro
    depends_on:
      - app1
      - app2
      - app3
      - app4
    healthcheck:
      test: ["CMD", "wget", "--quiet", "--tries=1", "--spider", "http://localhost/lb-health"]
      interval: 10s
      timeout: 5s
      retries: 3
    networks:
      - backend

volumes:
  postgres_data:

networks:
  backend:
    driver: bridge

    Networking Concepts Demonstrated

    1. Custom Bridge Network

    networks:
  backend:
    driver: bridge
    - Services can communicate using service names (DNS resolution) - Isolated from default bridge network - All services share same network

    2. Service Discovery

    • Service name resolves to an internal IP
    • Example: postgres resolves to PostgreSQL container's IP
    • Environment variables use service names: postgresql://app:app_password@postgres:5432/appdb

    3. Port Mapping

    # Format: "host_port:container_port"
ports:
  - "5001:5001"  # Make app1 accessible on host:5001
  - "5432:5432"  # Make postgres accessible on host:5432

    4. Health Checks & Dependency Management

    depends_on:
  postgres:
    condition: service_healthy
    - Ensures database is healthy before starting app - Not just waiting for port to be open - Check actual service readiness

    Usage

    Start the Stack

    # Start all services
docker-compose up -d

# View running services
docker-compose ps

# View logs
docker-compose logs
docker-compose logs -f app1  # Follow specific service
docker-compose logs -f       # Follow all services

    Scale Services

    # Scale to 6 app instances
docker-compose up -d --scale app=6

# Note: requires removing hardcoded container names

    Stop and Clean Up

    # Stop services (keep data)
docker-compose stop

# Remove containers
docker-compose down

# Remove containers AND volumes
docker-compose down -v

# Remove containers, volumes, and images
docker-compose down -v --rmi all

    Test Connectivity

    # Access service from within stack
docker-compose exec app1 curl http://app2:5002/health

# Access from host
curl http://localhost:80/api/v1/products

# Database from host
psql -h localhost -U app -d appdb

# Redis from host
redis-cli -h localhost

    Key Features

    Database Initialization

    • PostgreSQL creates user app with password app_password
    • Database appdb is automatically created
    • Health check ensures it's ready before apps start

    Multi-Instance Backend

    • Four Flask instances (app1-app4) on different ports
    • Each has unique INSTANCE_ID for tracking
    • Share same database and cache

    Load Balancing

    • Nginx distributes traffic using least_conn algorithm
    • Health checks on each backend instance
    • Automatic failover if instance goes down

    Caching Layer

    • Redis for distributed caching
    • Accessible from all Flask instances
    • Shared cache across instances

    Logging

    • All services output logs to stdout
    • View with docker-compose logs
    • Aggregation possible with logging drivers

    Networking Flows

    Request Flow

    Host:80 → Nginx:80 → app1:5001 (or app2, app3, app4)
               ↓
          PostgreSQL:5432
          Redis:6379

    Service Discovery Example

    app1 container resolves "postgres" hostname
  → Docker DNS server
  → Returns PostgreSQL container's IP (172.19.0.2)
  → Connection established on port 5432

    Port Accessibility

    From Host: - localhost:80 - Nginx - localhost:5001 - app1 - localhost:5432 - PostgreSQL

    From Containers: - nginx:80 - Service name (DNS) - app2:5002 - Service name - postgres:5432 - Service name

    Troubleshooting

    Service Won't Start

    # Check service logs
docker-compose logs postgres

# Check if port is in use
lsof -i :5432

# Restart specific service
docker-compose restart postgres

    Network Connectivity Issues

    # Inspect network
docker network ls
docker network inspect backend

# Test DNS resolution within stack
docker-compose exec app1 ping postgres

# Test port connectivity
docker-compose exec app1 curl http://postgres:5432

    Database Connection Issues

    # Check PostgreSQL logs
docker-compose logs postgres

# Connect to database
docker-compose exec postgres psql -U app -d appdb

# Check database exists
docker-compose exec postgres psql -U app -c "\l"

    Health Check Failures

    # Check individual health
docker-compose ps

# Manual health test
docker-compose exec app1 curl http://localhost:5001/health

# Check health check configuration
docker inspect app1 | grep -A 10 "Health"

    Production Adaptations

    For production use, consider:

    1. Database: Use managed database service (RDS, Cloud SQL)
    2. Secrets: Use proper secret management (Vault, AWS Secrets Manager)
    3. Scaling: Use Kubernetes instead of docker-compose
    4. Networking: Consider multiple networks for security
    5. Persistence: Mount volumes for data permanence
    6. Monitoring: Add ELK stack or similar for logs/metrics
    7. Security: Add authentication, authorization, encryption
    # Production adaptations example
services:
  app1:
    image: registry.example.com/app:v1.0  # Published image
    restart: always
    deploy:
      resources:
        limits:
          cpus: '1'
          memory: 512M
        reservations:
          cpus: '0.5'
          memory: 256M
    logging:
      driver: "json-file"
      options:
        max-size: "10m"
        max-file: "3"
