    Final Project: Enterprise Security Implementation

    Project Overview

    Design and implement a comprehensive security framework for a multi-tier, cloud-native application platform. This project integrates application security, infrastructure hardening, supply chain security, compliance automation, and incident response capabilities.

    Requirements

    Build an enterprise security posture that includes: - Secure SDLC with signed container images - Infrastructure security with policy enforcement - Network segmentation and zero-trust architecture - Secrets management and rotation - Compliance automation (SOC 2, PCI-DSS) - Container scanning and vulnerability management - Incident response automation - Security audit logging

    1.3 SAST (Static Application Security Testing)

    # security/sast-scanner.py
import subprocess
import json
import sys

class SASTScanner:
    def __init__(self, project_path):
        self.project_path = project_path

    def scan_with_semgrep(self):
        """Run Semgrep for SAST"""
        print("Running Semgrep SAST scan...")
        result = subprocess.run(
            ['semgrep', '--config=p/owasp-top-ten',
             '--json', self.project_path],
            capture_output=True,
            text=True
        )
        return json.loads(result.stdout)

    def scan_with_bandit(self):
        """Run Bandit for Python-specific checks"""
        print("Running Bandit security scan...")
        result = subprocess.run(
            ['bandit', '-r', '-f', 'json', self.project_path],
            capture_output=True,
            text=True
        )
        return json.loads(result.stdout)

    def scan_dependencies(self):
        """Scan for vulnerable dependencies"""
        print("Scanning dependencies...")
        result = subprocess.run(
            ['safety', 'check', '--json'],
            cwd=self.project_path,
            capture_output=True,
            text=True
        )
        return json.loads(result.stdout)

    def generate_report(self):
        """Generate comprehensive security report"""
        semgrep_issues = self.scan_with_semgrep()
        bandit_issues = self.scan_with_bandit()
        dependency_issues = self.scan_dependencies()

        report = {
            'semgrep': semgrep_issues,
            'bandit': bandit_issues,
            'dependencies': dependency_issues
        }

        # Fail build if critical issues found
        critical_count = sum(len(v['results']) for v in report.values())
        if critical_count > 0:
            print(f"FAIL: Found {critical_count} security issues")
            sys.exit(1)

        return report

if __name__ == '__main__':
    scanner = SASTScanner('.')
    report = scanner.generate_report()
    print(json.dumps(report, indent=2))

    2.3 Admission Controllers

    # security/admission-webhook.py
from flask import Flask, request, jsonify
import json
import base64

app = Flask(__name__)

@app.route('/validate', methods=['POST'])
def validate_pod():
    """Validate pod against security policies"""

    admission_review = request.get_json()
    pod = admission_review['request']['object']

    violations = []

    # Check: Images must be signed
    for container in pod['spec']['containers']:
        image = container['image']
        # In production, verify Cosign signature
        if not image.startswith('myregistry.azurecr.io'):
            violations.append(f"Untrusted registry: {image}")

    # Check: Resource limits must be set
    for container in pod['spec']['containers']:
        if 'resources' not in container or 'limits' not in container.get('resources', {}):
            violations.append(f"Container {container['name']} missing resource limits")

    # Check: Security context required
    for container in pod['spec']['containers']:
        sec_ctx = container.get('securityContext', {})
        if not sec_ctx.get('readOnlyRootFilesystem'):
            violations.append(f"Container {container['name']} not read-only")
        if sec_ctx.get('privileged'):
            violations.append(f"Container {container['name']} is privileged")

    # Generate admission response
    allowed = len(violations) == 0
    response = {
        "apiVersion": "admission.k8s.io/v1",
        "kind": "AdmissionReview",
        "response": {
            "uid": admission_review['request']['uid'],
            "allowed": allowed,
            "status": {
                "message": "; ".join(violations) if violations else "Approved"
            }
        }
    }

    return jsonify(response)

if __name__ == '__main__':
    app.run(ssl_context=('certs/tls.crt', 'certs/tls.key'))

    3.3 Automated Secrets Rotation

    # security/secrets-rotator.py
import os
import time
import hvac
import psycopg2
from datetime import datetime, timedelta

class SecretsRotator:
    def __init__(self, vault_addr, vault_token):
        self.client = hvac.Client(url=vault_addr, token=vault_token)

    def rotate_database_password(self, role_name):
        """Rotate database password"""
        print(f"Rotating database password for {role_name}...")

        try:
            # Request new credential from Vault
            creds = self.client.auth.kubernetes.read_static_role(
                path=f'db/static-creds/{role_name}',
                role=role_name
            )

            username = creds['data']['username']
            password = creds['data']['password']

            # Update application configuration
            self.update_app_config(username, password)

            # Verify connection works
            self.verify_connection(username, password)

            print(f"✓ Password rotated successfully at {datetime.now()}")

        except Exception as e:
            print(f"✗ Rotation failed: {e}")
            raise

    def rotate_api_keys(self, secret_path):
        """Rotate API keys"""
        print(f"Rotating API keys at {secret_path}...")

        new_api_key = self.client.secrets.kv_v2_read(secret_path)

        # Update all services using this key
        self.update_services_with_key(new_api_key)

        print(f"✓ API keys rotated")

    def update_app_config(self, username, password):
        """Update application config with new credentials"""
        config = {
            'DB_USER': username,
            'DB_PASSWORD': password,
            'DB_CONNECTION_REFRESHED': str(datetime.now())
        }
        # Update ConfigMap or restart pods
        os.environ.update(config)

    def verify_connection(self, username, password):
        """Verify database connection"""
        conn = psycopg2.connect(
            host=os.getenv('DB_HOST'),
            user=username,
            password=password,
            database=os.getenv('DB_NAME')
        )
        conn.close()

    def schedule_rotations(self):
        """Run continuous rotation schedule"""
        while True:
            try:
                self.rotate_database_password('myapp')
                self.rotate_api_keys('app/api-keys')
                # Rotate every 7 days
                time.sleep(7 * 24 * 60 * 60)
            except Exception as e:
                print(f"Rotation failed: {e}")
                time.sleep(60)  # Retry in 1 minute

if __name__ == '__main__':
    rotator = SecretsRotator(
        os.getenv('VAULT_ADDR'),
        os.getenv('VAULT_TOKEN')
    )
    rotator.schedule_rotations()

    4.2 CIS Kubernetes Benchmark Scan

    # compliance/cis-scanner.py
import subprocess
import json

class CISScanner:
    def __init__(self):
        self.results = {}

    def scan_with_kube_bench(self):
        """Run CIS benchmark scan"""
        print("Running CIS Kubernetes Benchmark...")

        result = subprocess.run(
            ['kube-bench', 'run', '--json'],
            capture_output=True,
            text=True
        )

        return json.loads(result.stdout)

    def check_rbac(self):
        """Verify RBAC configuration"""
        checks = {
            'anonymous_auth': self.verify_anonymous_disabled(),
            'api_server_cert': self.verify_tls_enabled(),
            'service_accounts': self.verify_service_accounts(),
            'network_policies': self.verify_network_policies()
        }
        return checks

    def verify_anonymous_disabled(self):
        """Check anonymous auth is disabled"""
        result = subprocess.run(
            ['kubectl', 'get', 'nodes', '-o', 'json'],
            capture_output=True,
            text=True
        )
        # Parse and verify --anonymous-auth=false
        return True

    def verify_tls_enabled(self):
        """Check TLS is enabled"""
        # Verify all API server certs
        return True

    def verify_service_accounts(self):
        """Check service account configuration"""
        result = subprocess.run(
            ['kubectl', 'get', 'sa', '-A', '-o', 'json'],
            capture_output=True,
            text=True
        )
        return json.loads(result.stdout)

    def verify_network_policies(self):
        """Check network policies are defined"""
        result = subprocess.run(
            ['kubectl', 'get', 'networkpolicies', '-A'],
            capture_output=True,
            text=True
        )
        return len(result.stdout.split('\n')) > 1

    def generate_compliance_report(self):
        """Generate compliance report"""
        cis_results = self.scan_with_kube_bench()
        rbac_checks = self.check_rbac()

        report = {
            'timestamp': str(datetime.now()),
            'cis_benchmark': cis_results,
            'rbac_configuration': rbac_checks,
            'overall_score': self.calculate_score(cis_results)
        }

        return report

    def calculate_score(self, results):
        """Calculate compliance score"""
        total = len(results['tests'])
        passed = sum(1 for t in results['tests'] if t['status'] == 'PASS')
        return (passed / total) * 100
