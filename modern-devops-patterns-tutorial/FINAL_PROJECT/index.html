    Final Project: Modern DevOps Patterns Integration

    Project Overview

    Integrate all three modern DevOps patterns (GitOps, Service Mesh, eBPF) into a single, production-ready Kubernetes platform. This project demonstrates mastery of declarative infrastructure, advanced networking, and kernel-level observability.

    Requirements

    Build a comprehensive DevOps infrastructure that includes: - GitOps-based deployment with ArgoCD - Service Mesh with Istio for traffic control - eBPF-based observability with Cilium - Multi-cluster management - Automated canary deployments - Real-time kernel monitoring - Zero-downtime updates

    System Architecture

    Git Repository (Source of Truth)
    ↓
GitOps Controller (ArgoCD)
    ├─ Watches for changes
    ├─ Syncs to clusters
    └─ Triggers deployments
         ↓
    ┌─────────────────────────────────┐
    │  Kubernetes Cluster             │
    ├─────────────────────────────────┤
    │                                 │
    │  Service Mesh Layer (Istio)     │
    │  ├─ VirtualServices             │
    │  ├─ DestinationRules            │
    │  ├─ Canary deployments          │
    │  └─ Circuit breaking            │
    │           ↕                      │
    │  Kernel Layer (eBPF + Cilium)   │
    │  ├─ Cilium CNI                  │
    │  ├─ Network policies            │
    │  ├─ Service mesh acceleration   │
    │  └─ Real-time monitoring        │
    │                                 │
    │  Applications                   │
    │  ├─ Frontend                    │
    │  ├─ Backend APIs                │
    │  ├─ Microservices               │
    │  └─ Data pipeline               │
    │                                 │
    └─────────────────────────────────┘
         ↕
    Observability Stack
    ├─ Prometheus (metrics from eBPF)
    ├─ Loki (logs)
    ├─ Grafana (visualization)
    └─ Jaeger (tracing via Istio)

    3.3 Real-Time Monitoring with Cilium/Hubble

    # observability/cilium-monitor.py
import json
import subprocess
import time
from collections import defaultdict

class CiliumMonitor:
    def __init__(self):
        self.metrics = defaultdict(int)

    def get_ebpf_metrics(self):
        """Get eBPF-based network metrics"""
        try:
            # Query dropped packets
            result = subprocess.run(
                ['cilium', 'bpf', 'metrics', 'list'],
                capture_output=True,
                text=True
            )

            metrics = {}
            for line in result.stdout.split('\n'):
                if line.strip():
                    parts = line.split(':')
                    if len(parts) == 2:
                        metric_name, value = parts
                        metrics[metric_name.strip()] = int(value.strip())

            return metrics
        except Exception as e:
            print(f"Error getting metrics: {e}")
            return {}

    def get_traffic_distribution(self):
        """Get traffic distribution using Hubble"""
        try:
            result = subprocess.run(
                ['hubble', 'observe', '--json'],
                capture_output=True,
                text=True
            )

            traffic = defaultdict(int)
            for line in result.stdout.split('\n'):
                if line.strip():
                    flow = json.loads(line)
                    source = flow['source']['namespace']
                    dest = flow['destination']['namespace']
                    traffic[f'{source} -> {dest}'] += 1

            return dict(traffic)
        except Exception as e:
            print(f"Error getting traffic: {e}")
            return {}

    def monitor_loop(self, interval=10):
        """Continuous monitoring loop"""
        while True:
            metrics = self.get_ebpf_metrics()
            traffic = self.get_traffic_distribution()

            print(f"\n=== eBPF Metrics ({time.ctime()}) ===")
            for metric, value in metrics.items():
                print(f"{metric}: {value}")

            print(f"\n=== Traffic Distribution ===")
            for conn, count in traffic.items():
                print(f"{conn}: {count} flows")

            time.sleep(interval)

if __name__ == '__main__':
    monitor = CiliumMonitor()
    monitor.monitor_loop()

    4.1 End-to-End Deployment

    #!/bin/bash
set -e

REPO="https://github.com/myorg/devops-platform"
ENVIRONMENT=${1:-staging}

echo "Deploying to $ENVIRONMENT..."

# 1. Add GitOps repository
argocd repo add $REPO --username $GIT_USER --password $GIT_TOKEN

# 2. Create ArgoCD Application
cat <<EOF | kubectl apply -f -
apiVersion: argoproj.io/v1alpha1
kind: Application
metadata:
  name: $ENVIRONMENT-app
  namespace: argocd
spec:
  project: default
  source:
    repoURL: $REPO
    targetRevision: main
    path: overlays/$ENVIRONMENT
  destination:
    server: https://kubernetes.default.svc
    namespace: $ENVIRONMENT
  syncPolicy:
    automated:
      prune: true
      selfHeal: true

# 3. Wait for sync
echo "Waiting for sync..."
kubectl wait --for=condition=Synced \
  application/$ENVIRONMENT-app \
  -n argocd \
  --timeout=5m

# 4. Verify deployment
echo "Verifying deployment..."
kubectl rollout status deployment/backend-api -n $ENVIRONMENT
kubectl rollout status deployment/frontend -n $ENVIRONMENT

echo "✓ Deployment successful!"

    4.2 Canary Analysis

    #!/bin/bash

# Monitor canary metrics
DURATION=300  # 5 minutes
INTERVAL=10

echo "Monitoring canary deployments..."

for ((i=0; i<$DURATION; i+=$INTERVAL)); do
  ERROR_RATE=$(kubectl get vs backend-vs -o jsonpath='{.spec.http[0].route[1].weight}' || echo "unknown")
  LATENCY=$(kubectl exec -it $(kubectl get pods -l app=backend-api -o name | head -1) \
    -- curl -s http://localhost:8000/metrics | grep request_duration_seconds_total)

  echo "[$i/$DURATION] Error rate: $ERROR_RATE%, Latency: $LATENCY"

  sleep $INTERVAL
done

# Promote canary if metrics are good
echo "Promoting canary to stable..."
kubectl patch vs backend-vs -p '{"spec":{"http":[{"route":[{"destination":{"host":"backend-api","subset":"stable"},"weight":100}]}]}}' --type=merge

    5.2 Automated Rollback

    # automation/auto-rollback.py
import subprocess
import time
from prometheus_client import Prometheus

class AutoRollback:
    def __init__(self, prometheus_url, app_name):
        self.prometheus = Prometheus(prometheus_url)
        self.app_name = app_name
        self.error_threshold = 5.0  # 5% error rate

    def get_error_rate(self, minutes=5):
        """Get error rate from Prometheus"""
        query = f'rate(requests_total{{status=~"5.."}}[{minutes}m]) / rate(requests_total[{minutes}m]) * 100'
        result = self.prometheus.query(query)
        return float(result[0]['value'][1]) if result else 0

    def rollback(self):
        """Trigger automatic rollback"""
        print(f"Rolling back {self.app_name}...")
        subprocess.run([
            'kubectl', 'rollout', 'undo',
            f'deployment/{self.app_name}',
            '-n', 'production'
        ])

    def monitor(self):
        """Continuous monitoring with automatic rollback"""
        while True:
            error_rate = self.get_error_rate()
            if error_rate > self.error_threshold:
                print(f"ERROR RATE TOO HIGH: {error_rate}%")
                self.rollback()
            time.sleep(30)
