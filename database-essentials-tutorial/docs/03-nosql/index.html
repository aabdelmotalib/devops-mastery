    Introduction & Overview

    The NoSQL Philosophy

    NoSQL databases solve different problems than relational databases:

    Aspect SQL (PostgreSQL) NoSQL (MongoDB, Redis)
    Schema Fixed, enforced Flexible, dynamic
    Consistency ACID transactions Eventually consistent (usually)
    Scaling Vertical (bigger server) Horizontal (more servers)
    Joins Built-in, powerful No joins (data embedding/duplication)
    Data Model Tables & rows Documents, key-value, graphs
    Query Language SQL Database-specific (MongoDB: aggregation)
    Best For Transactional data, complex relationships High-volume writes, flexible schemas, caching

    Key Principle

    Don't replace PostgreSQL with NoSQL. Complement it.

    A robust production system uses PostgreSQL for core transactional data and NoSQL for specific problems it solves better.

    What You'll Learn

    • MongoDB: Document storage for flexible schemas, logging, catalogs
    • Redis: In-memory caching, sessions, real-time data
    • Architecture patterns: Combining SQL + NoSQL effectively
    • Production-ready code: Monitoring, error handling, scaling
    • Common mistakes: Anti-patterns to avoid

    MongoDB: Document Database

    What is MongoDB?

    MongoDB stores data as JSON-like documents (BSON internally) rather than tables and rows. Each document can have a different structure, making it ideal for evolving schemas.

    Visual Comparison: SQL vs Document Model

    PostgreSQL (Structured):
┌─ users table ───────────────────────┐
│ id │ email              │ age │ name  │
├────┼────────────────────┼─────┼───────┤
│ 1  │ john@example.com   │ 30  │ John  │
│ 2  │ jane@example.com   │ 28  │ Jane  │
└─────────────────────────────────────┘

MongoDB (Flexible):
{
  "_id": ObjectId("..."),
  "email": "john@example.com",
  "age": 30,
  "name": "John",
  "skills": ["Python", "Kubernetes"],    ← Different structure
  "verified": true
}

{
  "_id": ObjectId("..."),
  "email": "jane@example.com",
  "age": 28,
  "name": "Jane",
  "certifications": [{name: "CKA", year: 2023}],  ← Can differ
  "verified": false,
  "phone": "+1234567890"  ← Additional field, not in other docs
}

    Core Characteristics

    Feature Advantage Trade-off
    Flexible Schema Add fields without migration Risk of inconsistent data
    Nested Documents No joins needed, single query Data duplication, update complexity
    Horizontal Scaling Handle massive data volumes Sharding complexity
    Eventually Consistent Faster writes by default Stale reads possible
    Rich Documents Store arrays, objects, dates Larger documents = more memory

    Document Structure

    Basic Example

    {
  "_id": ObjectId("507f1f77bcf86cd799439011"),  // Auto-generated ID
  "email": "user@example.com",

  // Nested object (can be queried)
  "profile": {
    "firstName": "John",
    "lastName": "Doe",
    "age": 30,
    "location": "New York"
  },

  // Array of values
  "skills": ["Python", "Kubernetes", "DevOps"],

  // Array of objects
  "experience": [
    {
      "company": "TechCorp",
      "role": "DevOps Engineer",
      "years": 5
    },
    {
      "company": "CloudInc",
      "role": "SRE",
      "years": 2
    }
  ],

  // Standard types
  "isActive": true,
  "joinedAt": ISODate("2023-01-15T10:30:00Z"),
  "salary": 120000
}

    Design Guidelines

    Document Size: MongoDB has a 16MB limit per document Nesting: Usually 2-3 levels deep maximum Arrays: Unbounded arrays should reference instead (see patterns below)

# 3. Install MongoDB

    Python Setup

    # backend/app/db/mongo.py
from pymongo import MongoClient
from pymongo.errors import DuplicateKeyError, ConnectionFailure
import os

class MongoDatabase:
    def __init__(self):
        """Initialize MongoDB connection."""
        uri = os.getenv('MONGO_URI', 'mongodb://localhost:27017/')
        try:
            self.client = MongoClient(uri, serverSelectionTimeoutMS=5000)
            # Verify connection
            self.client.admin.command('ping')
            self.db = self.client[os.getenv('MONGO_DB', 'devops_db')]
        except ConnectionFailure as e:
            raise RuntimeError(f"Failed to connect to MongoDB: {e}")

    def get_collection(self, collection_name):
        """Get a collection object."""
        return self.db[collection_name]

    def close(self):
        """Close the connection."""
        self.client.close()

# Singleton instance
mongo_db = MongoDatabase()

    Basic CRUD Operations

    Create (Insert)

    from app.db.mongo import mongo_db
from datetime import datetime

collection = mongo_db.get_collection('users')

# Insert single document
user_doc = {
    "email": "john@example.com",
    "name": "John Doe",
    "role": "engineer",
    "created_at": datetime.utcnow()
}
result = collection.insert_one(user_doc)
print(f"Inserted ID: {result.inserted_id}")

# Insert multiple documents
users = [
    {"email": "jane@example.com", "name": "Jane Doe"},
    {"email": "bob@example.com", "name": "Bob Smith"}
]
result = collection.insert_many(users)
print(f"Inserted {len(result.inserted_ids)} documents")

    Read (Query)

    # Find single document
user = collection.find_one({"email": "john@example.com"})

# Find by ID
from bson import ObjectId
user = collection.find_one({"_id": ObjectId("507f1f77bcf86cd799439011")})

# Find multiple documents
all_users = collection.find()
for user in all_users:
    print(user['email'])

# Query with filters
engineers = collection.find({"role": "engineer"})
active_users = collection.find({"isActive": True})

# Query nested fields
users_in_nyc = collection.find({"profile.location": "New York"})

# Query arrays
python_devs = collection.find({"skills": "Python"})

    Update

    # Update single document
collection.update_one(
    {"email": "john@example.com"},
    {"$set": {"role": "senior engineer", "updated_at": datetime.utcnow()}}
)

# Update multiple documents
result = collection.update_many(
    {"role": "engineer"},
    {"$set": {"department": "Platform"}}
)
print(f"Modified {result.modified_count} documents")

# Array operations
collection.update_one(
    {"email": "john@example.com"},
    {"$push": {"skills": "Docker"}}  # Add to array
)

collection.update_one(
    {"email": "john@example.com"},
    {"$pull": {"skills": "Python"}}  # Remove from array
)

    Delete

    # Delete single document
result = collection.delete_one({"email": "john@example.com"})

# Delete multiple documents
result = collection.delete_many({"role": "intern"})
print(f"Deleted {result.deleted_count} documents")

# Delete all documents (use carefully!)
collection.delete_many({})

    Indexing for Performance

    Indexes are crucial for MongoDB query performance:

    collection = mongo_db.get_collection('users')

# Single field index
collection.create_index("email", unique=True)
collection.create_index("created_at")

# Compound index (multiple fields)
collection.create_index([("role", 1), ("created_at", -1)])
# 1 = ascending, -1 = descending

# Text search index (full-text search)
collection.create_index([("bio", "text"), ("skills", "text")])

# TTL index (auto-delete documents after time)
collection.create_index("created_at", expireAfterSeconds=2592000)  # 30 days

# List all indexes
collection.list_indexes()

# Drop an index
collection.drop_index("email_1")

    MongoDB Schema Patterns

    Pattern 1: Embedding (Denormalization)

    Use when related data is always accessed together:

    # Good: Address embedded in User
user = {
    "_id": ObjectId("..."),
    "email": "user@example.com",
    "addresses": [
        {
            "type": "shipping",
            "street": "123 Main St",
            "city": "NYC",
            "zip": "10001"
        },
        {
            "type": "billing",
            "street": "456 Secondary Ave",
            "city": "LA",
            "zip": "90001"
        }
    ]
}

# Query embedded field
users_in_nyc = collection.find({"addresses.city": "NYC"})

    When to use embedding: - Data accessed together - 1-to-few relationships - Limited array size - Data doesn't change frequency

    Pattern 2: Referencing (Normalization)

    Use when data is accessed independently:

    # User document
{
    "_id": ObjectId("user123"),
    "email": "user@example.com",
    "name": "John Doe"
}

# Order document (references user)
{
    "_id": ObjectId("order456"),
    "user_id": ObjectId("user123"),  # Reference, not embedding
    "total": 99.99,
    "items": [
        {"product_id": "prod1", "quantity": 2}
    ]
}

# Query: Find user's orders
orders = orders_collection.find({"user_id": ObjectId("user123")})

# Note: MongoDB doesn't do automatic joins, you must manually fetch
user = users_collection.find_one({"_id": ObjectId("user123")})
orders = orders_collection.find({"user_id": user['_id']})

    When to use referencing: - Large or unbounded arrays - Data accessed separately
    - Many-to-many relationships - Frequent updates

    Real-World MongoDB Examples

    Example 1: Application Logging

    # backend/app/repositories/log_repository.py
from app.db.mongo import mongo_db
from datetime import datetime, timedelta
from enum import Enum

class LogLevel(Enum):
    INFO = "INFO"
    WARNING = "WARNING"
    ERROR = "ERROR"
    DEBUG = "DEBUG"

class LogRepository:
    def __init__(self):
        self.collection = mongo_db.get_collection('logs')
        # Create indexes for common queries
        self.collection.create_index('timestamp')
        self.collection.create_index('level')
        self.collection.create_index('service')
        # Auto-delete logs after 30 days
        self.collection.create_index('timestamp', expireAfterSeconds=2592000)

    def insert_log(self, level: LogLevel, message: str, service: str, 
                   metadata: dict = None):
        """Insert a log entry."""
        doc = {
            'level': level.value,
            'message': message,
            'service': service,
            'metadata': metadata or {},
            'timestamp': datetime.utcnow()
        }
        result = self.collection.insert_one(doc)
        return str(result.inserted_id)

    def get_logs(self, level: LogLevel = None, service: str = None, 
                 hours: int = 24, limit: int = 100):
        """Retrieve logs with optional filtering."""
        query = {}

        if level:
            query['level'] = level.value
        if service:
            query['service'] = service

        # Filter by time range
        cutoff = datetime.utcnow() - timedelta(hours=hours)
        query['timestamp'] = {'$gte': cutoff}

        return list(self.collection\
            .find(query)\
            .sort('timestamp', -1)\
            .limit(limit))

    def get_error_summary(self, hours: int = 24):
        """Get count of errors by service."""
        cutoff = datetime.utcnow() - timedelta(hours=hours)

        # Aggregation pipeline
        pipeline = [
            {
                '$match': {
                    'level': 'ERROR',
                    'timestamp': {'$gte': cutoff}
                }
            },
            {
                '$group': {
                    '_id': '$service',
                    'count': {'$sum': 1}
                }
            },
            {
                '$sort': {'count': -1}
            }
        ]

        return list(self.collection.aggregate(pipeline))

    Example 2: Product Catalog

    # backend/app/repositories/product_repository.py
from app.db.mongo import mongo_db
from datetime import datetime

class ProductRepository:
    def __init__(self):
        self.collection = mongo_db.get_collection('products')
        # Create indexes
        self.collection.create_index('sku', unique=True)
        self.collection.create_index('category')
        self.collection.create_index([('name', 'text'), ('description', 'text')])

    def create_product(self, sku: str, name: str, price: float, 
                      category: str, attributes: dict = None):
        """Create a new product."""
        doc = {
            'sku': sku,
            'name': name,
            'price': price,
            'category': category,
            'attributes': attributes or {},
            'inventory': 0,
            'created_at': datetime.utcnow(),
            'updated_at': datetime.utcnow()
        }
        result = self.collection.insert_one(doc)
        return str(result.inserted_id)

    def get_product_by_sku(self, sku: str):
        """Get product by SKU."""
        return self.collection.find_one({'sku': sku})

    def search_products(self, query: str, category: str = None, limit: int = 20):
        """Full-text search products."""
        search_filter = {'$text': {'$search': query}}

        if category:
            search_filter['category'] = category

        return list(self.collection\
            .find(search_filter)\
            .limit(limit))

    def get_by_category(self, category: str):
        """Get all products in a category."""
        return list(self.collection.find({'category': category}))

    def update_price(self, sku: str, new_price: float):
        """Update product price."""
        self.collection.update_one(
            {'sku': sku},
            {'$set': {'price': new_price, 'updated_at': datetime.utcnow()}}
        )

    def update_inventory(self, sku: str, quantity_change: int):
        """Increment or decrement inventory."""
        self.collection.update_one(
            {'sku': sku},
            {'$inc': {'inventory': quantity_change}}
        )

    MongoDB Anti-Patterns (What Not To Do)

    ❌ Anti-Pattern 1: Using for Transactional Data

    # WRONG: User authentication in MongoDB
def register_user(email: str, password: str):
    users = mongo_db.get_collection('users')

    # Problems:
    # - No transaction guarantees
    # - No foreign key constraints
    # - Race conditions possible
    # - Cannot rollback on error

    users.insert_one({
        'email': email,
        'password_hash': hash_password(password)
    })

    Solution: Use PostgreSQL for user authentication

    ❌ Anti-Pattern 2: Unbounded Array Growth

    # WRONG: Embedding all orders in user document
{
    "_id": ObjectId("user123"),
    "email": "user@example.com",
    "orders": [
        {order1},
        {order2},
        ...
        {order1000000}  # Document grows forever!
    ]
}

# Problems:
# - Violates 16MB document limit
# - Entire document loaded into memory
# - Slow queries

    Solution: Reference instead of embedding

    # RIGHT: Reference orders separately
# users collection
{
    "_id": ObjectId("user123"),
    "email": "user@example.com"
}

# orders collection
{
    "_id": ObjectId("order1"),
    "user_id": ObjectId("user123"),
    "total": 99.99
}

    ❌ Anti-Pattern 3: Ignoring Indexes

    # WRONG: Querying without indexes
users.find({"email": "user@example.com"})  # Full collection scan!

# Solution: Create indexes
users.create_index("email", unique=True)

    Redis: In-Memory Data Store

    What is Redis?

    Redis is an in-memory key-value store that's extremely fast, built for caching, sessions, and real-time data. It's not a database replacement—it's a complement to your primary database.

    Redis vs Databases

    Aspect Redis PostgreSQL
    Storage RAM (all data) Disk + memory
    Speed ~0.1ms ~1-5ms
    Data Types Strings, Lists, Sets, Sorted Sets, Hashes Tables, Complex types
    Persistence Optional (RDB/AOF) Default
    Transactions Single-threaded atomicity Full ACID
    Use Case Caching, sessions, real-time Persistent storage

    Performance Comparison

    Operation         | PostgreSQL | Redis
──────────────────────────────────────────
Simple read       | 1-5ms      | 0.05-0.1ms (10-100x faster)
Sequential reads  | 1-5ms each | 0.05-0.1ms each
Complex query     | 10-100ms   | N/A (no SQL)
Memory usage/GB   | Scalable   | All in RAM (expensive)
Data size         | TB+ easy   | Limited by RAM

    Data Structures

    Redis offers more than key-value pairs:

    1. Strings (Simple Values)

from app.db.redis import redis_client
from flask import request, jsonify
from functools import wraps

def rate_limit(max_requests: int = 100, window_seconds: int = 60):
    """Rate limit decorator by IP address."""
    def decorator(f):
        @wraps(f)
        def wrapped(*args, **kwargs):
            # Use IP as key
            ip = request.remote_addr
            key = f'rate:{ip}:{f.__name__}'

            # Increment request counter
            current = redis_client.incr(key)

            # Set expiration on first request
            if current == 1:
                redis_client.expire(key, window_seconds)

            # Check if exceeded limit
            if current > max_requests:
                retry_after = redis_client.ttl(key)
                return jsonify({
                    'error': 'Rate limit exceeded',
                    'retry_after_seconds': retry_after
                }), 429

            return f(*args, **kwargs)
        return wrapped
    return decorator

# Usage in Flask app

class EmailQueue:
    QUEUE_KEY = 'queue:emails'

    def enqueue(self, to: str, subject: str, body: str,  priority: int = 1):
        """Add email to queue."""
        email_data = {
            'to': to,
            'subject': subject,
            'body': body,
            'priority': priority,
            'queued_at': datetime.utcnow().isoformat()
        }

        # Use sorted set for priority
        score = -priority  # Negative so lower numbers = higher priority
        redis_client.client.zadd(
            self.QUEUE_KEY,
            {json.dumps(email_data): score}
        )

    def dequeue(self):
        """Get next email from queue."""
        # Get highest priority (lowest score)
        data = redis_client.client.zrange(self.QUEUE_KEY, 0, 0)
        if data:
            email = json.loads(data[0])
            redis_client.client.zrem(self.QUEUE_KEY, data[0])
            return email
        return None

    def queue_size(self):
        """Get pending email count."""
        return redis_client.client.zcard(self.QUEUE_KEY)

from app.db.postgres import db
from app.db.redis import redis_client
import json

class ReadThroughCache:
    """
    Cache layer between app and database.
    App always reads from cache.
    Cache handles database interaction.
    """

    def __init__(self, namespace: str, ttl: int = 300):
        self.namespace = namespace
        self.ttl = ttl

    def get(self, key: str, loader_func):
        """
        Get from cache or load from database.

        Args:
            key: Cache key
            loader_func: Function to call if cache miss

        Returns:
            Cached or loaded data
        """
        cache_key = f'{self.namespace}:{key}'

        # Try cache
        cached = redis_client.get(cache_key)
        if cached:
            return json.loads(cached)

        # Cache miss: load from database
        data = loader_func()

        # Store result
        if data:
            redis_client.set(
                cache_key,
                json.dumps(data),
                ex=self.ttl
            )

        return data

    def invalidate(self, key: str):
        """Remove from cache."""
        redis_client.delete(f'{self.namespace}:{key}')

# Usage
user_cache = ReadThroughCache('user', ttl=300)

def get_user(user_id):
    return user_cache.get(
        f'id:{user_id}',
        loader_func=lambda: User.query.get(user_id).to_dict()
    )

def update_user(user_id, **kwargs):
    user = User.query.get(user_id)
    for k, v in kwargs.items():
        setattr(user, k, v)
    db.session.commit()

    user_cache.invalidate(f'id:{user_id}')  # Invalidate cache

class WriteBehindCache:
    """
    Write to cache immediately, flush to DB periodically.
    Fast writes, eventual consistency.
    """

    def __init__(self, namespace: str, flush_interval: int = 60):
        self.namespace = namespace
        self.flush_interval = flush_interval

        redis_client.set(cache_key, json.dumps(data))

    def flush_to_db(self):
        """
        Periodically write cache to database.
        Run as background task.
        """
        keys = redis_client.client.keys(f'{self.namespace}:*')
        for key in keys:
            data = json.loads(redis_client.get(key))
            # Write to database
            # Clear from cache after flush
            redis_client.delete(key)

def handle_mongo_error(func):
    """Decorator for MongoDB error handling."""
    def wrapper(*args, **kwargs):
        try:
            return func(*args, **kwargs)
        except mongo_errors.DuplicateKeyError:

    def wrapper(*args, **kwargs):

        except redis.ConnectionError:
            # Fall back to database query
            return None
        except redis.TimeoutError:
            # Timeout on Redis, don't crash
            return None
    return wrapper

    When to Use Each

    Decision Tree

    Do you need flexible schema?
├─ YES: Document data with arrays/objects?
│   ├─ YES: Use MongoDB
│   └─ NO: Use PostgreSQL (JSONB)
└─ NO: Use PostgreSQL

Need microsecond latency caching?
├─ YES: Use Redis
└─ NO: Use database

Need to store sessions/temporary data?
├─ YES: Use Redis
└─ NO: Use database

Need full-text search?
├─ YES: MongoDB (moderate scale) or Elasticsearch (large scale)
└─ NO: Use PostgreSQL (text search)

Need real-time leaderboard/ranking?
├─ YES: Use Redis (sorted sets)
└─ NO: Use database with indexes

    Use MongoDB For:

    • ✅ Logging - Flexible schema, high write volume, time-windowed queries
    • ✅ Product catalogs - Varying attributes per product
    • ✅ Content management - Articles with different structures
    • ✅ Analytics events - High-volume, flexible structure
    • ✅ Cache of API responses - Fast read, can be recreated

    DO NOT Use MongoDB For:

    • ❌ User authentication - Needs ACID guarantees
    • ❌ Financial transactions - Needs constraints and rollback
    • ❌ Data with complex relationships - Avoid denormalization
    • ❌ Data requiring joins - SQL is simpler
    • ❌ Anything with integrity constraints - PostgreSQL enforces at database level

    Use Redis For:

    • ✅ Caching - Reduce database load
    • ✅ Sessions - Fast reads, auto-expiration
    • ✅ Rate limiting - Atomic counters
    • ✅ Leaderboards - Sorted sets, fast ranking
    • ✅ Real-time counters - Page views, metrics
    • ✅ Task queues - Inter-process communication
    • ✅ Pub/Sub messaging - Real-time notifications

    DO NOT Use Redis For:

    • ❌ Persistent storage - Data lost on restart
    • ❌ Large datasets - Limited by RAM
    • ❌ Complex queries - No SQL
    • ❌ Unstructured data - No schema validation

    Redis: In-Memory Data Store

    What Redis Actually Is

    Redis is an in-memory key-value store that's extremely fast. It's not a replacement for databases—it's a complement. Think of it as a turbocharger for your application.

    How it works: - Data stored in RAM (memory) - 100-1000x faster than disk databases - Single-threaded - No race conditions, atomic operations - Optional persistence - Can save to disk, or lose on restart - Data structures - Not just strings, but lists, sets, sorted sets, hashes

    Performance comparison: 

    Operation         | PostgreSQL | Redis
─────────────────────────────────────────
Simple read       | 1-5ms      | 0.01-0.1ms (100-500x faster)
Write             | 1-5ms      | 0.01-0.1ms
Complex query     | 10-100ms   | N/A (no SQL)
Memory usage      | On disk    | All in RAM (expensive)

    Data structures (why Redis is powerful):

    import redis

r = redis.Redis(host='localhost', port=6379)

# 1. String - Simple value
r.set('user:1:name', 'John')              # SET
r.get('user:1:name')                      # GET → 'John'

# 2. List - Ordered collection (like array)
r.lpush('queue:jobs', job1, job2, job3)  # Add to left (push)
r.rpop('queue:jobs')                      # Remove from right (pop)

# 3. Set - Unordered unique collection
r.sadd('tags:python', 'user1', 'user2')  # Add tag
r.smembers('tags:python')                 # Get all users with tag
r.sismember('tags:python', 'user1')       # Check membership

# 4. Sorted Set - Ordered by score
r.zadd('leaderboard', {'user1': 100, 'user2': 95})  # Score is ranking
r.zrange('leaderboard', 0, -1, withscores=True)     # Top to bottom

# 5. Hash - Dictionary/object
r.hset('user:1', mapping={                # SET multiple fields
    'name': 'John',
    'email': 'john@example.com',
    'age': 30
})
r.hget('user:1', 'name')                  # GET single field → 'John'
r.hgetall('user:1')                       # GET all fields

    Redis Persistence Options

    Redis can optionally save to disk so data survives restarts:

    # In Redis configuration file (/etc/redis/redis.conf):

# Option 1: RDB (Snapshot)
# Saves entire dataset to disk periodically
save 900 1      # Save if 1 key changed in 900 seconds
save 300 10     # Save if 10 keys changed in 300 seconds
save 60 10000   # Save if 10000 keys changed in 60 seconds

# Option 2: AOF (Append-Only File)
# Logs every write operation
appendonly yes
appendfsync everysec  # Sync to disk every second

# Option 3: No persistence (fastest, loses data on restart)
# Don't set any save directives

    When to enable persistence: - Yes: Session data, important cache data - No: Throwaway cache, rate limit counters, ephemeral data

from app.db.postgres import db
from app.db.redis import redis_client
from app.models import User
import json

class UserService:
    CACHE_TTL = 300  # 5 minutes

    def get_user(self, user_id):
        # Try cache first
        cache_key = f'user:{user_id}'
        cached = redis_client.get(cache_key)

        if cached:
            return json.loads(cached)

        # Cache miss - query database
        user = User.query.get(user_id)
        if not user:
            return None

        user_data = user.to_dict()

        # Store in cache
        redis_client.set(
            cache_key,
            json.dumps(user_data),
            ex=self.CACHE_TTL
        )

        return user_data

    def update_user(self, user_id, **kwargs):
        user = User.query.get(user_id)
        for key, value in kwargs.items():
            setattr(user, key, value)
        db.session.commit()

        # Invalidate cache
        redis_client.delete(f'user:{user_id}')

        return user.to_dict()

    Rate Limiting

    # backend/app/middleware/rate_limit.py
from app.db.redis import redis_client
from flask import request, jsonify
from functools import wraps

def rate_limit(max_requests=100, window=60):
    """
    Rate limit decorator
    max_requests: Maximum requests allowed
    window: Time window in seconds
    """
    def decorator(f):
        @wraps(f)
        def wrapped(*args, **kwargs):
            # Use IP address as key
            ip = request.remote_addr
            key = f'rate_limit:{ip}:{f.__name__}'

            # Increment counter
            current = redis_client.client.incr(key)

            # Set expiration on first request
            if current == 1:
                redis_client.client.expire(key, window)

            # Check limit
            if current > max_requests:
                return jsonify({
                    'error': 'Rate limit exceeded',
                    'retry_after': redis_client.client.ttl(key)
                }), 429

            return f(*args, **kwargs)
        return wrapped
    return decorator

# Usage

def expensive_operation():
    return jsonify({'result': 'success'})

class EmailQueue:
    QUEUE_KEY = 'queue:emails'

    def enqueue(self, to, subject, body):
        email_data = {
            'to': to,
            'subject': subject,
            'body': body,
            'queued_at': datetime.utcnow().isoformat()
        }

        data = redis_client.client.rpop(self.QUEUE_KEY)
        return json.loads(data) if data else None

    def queue_size(self):
        return redis_client.client.llen(self.QUEUE_KEY)
